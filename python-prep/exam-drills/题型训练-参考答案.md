## A1 ParkingSystem

```python
from typing import List

class ParkingSystem:
    def __init__(self, slots: int):
        self.slots = slots
        self.cars = set()

    def park(self, car_id: int) -> int:
        if car_id in self.cars or len(self.cars) >= self.slots:
            return -1
        self.cars.add(car_id)
        return len(self.cars)

    def leave(self, car_id: int) -> int:
        if car_id not in self.cars:
            return -1
        self.cars.remove(car_id)
        return len(self.cars)

    def status(self) -> List[int]:
        return sorted(self.cars)
```

## A2 RateLimiter

```python
from typing import List
from collections import defaultdict, deque

class RateLimiter:
    def __init__(self, window: int, limit: int):
        self.window = window
        self.limit = limit
        self.ok = defaultdict(deque)  # user -> timestamps of accepted reqs
        self.seen = set()

    def request(self, t: int, user_id: int) -> bool:
        q = self.ok[user_id]
        while q and q[0] <= t - self.window:
            q.popleft()
        if len(q) >= self.limit:
            return False
        q.append(t)
        self.seen.add(user_id)
        return True

    def active_users(self) -> List[int]:
        return sorted(self.seen)
```

```python
def shortest_good(good, need):
    n = len(good)
    left = 0
    cnt = 0
    ans = n + 1
    for right, g in enumerate(good):
        cnt += g
        while cnt >= need:
            ans = min(ans, right - left + 1)
            cnt -= good[left]
            left += 1
    return -1 if ans == n + 1 else ans
```

## C1 exec_mini

```python
from typing import List

def exec_mini(instructions: List[str], mem: List[int]) -> List[int]:
    regs = [0] * 8
    stack = []
    pc = 0
    steps = 0
    mem = list(mem)

    while 0 <= pc < len(instructions) and steps < 10000:
        steps += 1
        raw = instructions[pc]
        opcode = int(raw[0:2], 16)
        reg = int(raw[2:4], 16)
        imm = int(raw[4:6], 16)

        if opcode == 0x01:  # LOAD
            if 0 <= imm < len(mem) and 0 <= reg < 8:
                regs[reg] = mem[imm]
            pc += 1
        elif opcode == 0x02:  # STORE
            if 0 <= imm < len(mem) and 0 <= reg < 8:
                mem[imm] = regs[reg]
            pc += 1
        elif opcode == 0x03:  # ADD
            if 0 <= reg < 8:
                regs[reg] = (regs[reg] + imm) % 65536
            pc += 1
        elif opcode == 0x04:  # JMP
            pc = imm
        elif opcode == 0x05:  # PUSH
            if 0 <= reg < 8:
                stack.append(regs[reg])
            pc += 1
        elif opcode == 0x06:  # POP
            if 0 <= reg < 8:
                regs[reg] = stack.pop() if stack else 0
            pc += 1
        elif opcode == 0x07:  # HALT
            break
        else:
            pc += 1

    return mem
```
